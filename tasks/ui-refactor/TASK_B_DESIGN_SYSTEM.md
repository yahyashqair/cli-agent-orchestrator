export function applyTheme(theme: Theme) {
  const root = document.documentElement
  const themeConfig = themes[theme]

  Object.entries(themeConfig.colors).forEach(([key, value]) => {
    root.style.setProperty(`--${key.replace(/([A-Z])/g, '-$1').toLowerCase()}`, value)
  })

  root.setAttribute('data-theme', theme)
}

### 7. Create Custom Hooks
Create `src/hooks/use-theme.ts`:
```typescript
import { useState, useEffect } from 'react'
import { Theme, applyTheme } from '@/lib/theme'

export function useTheme() {
  const [theme, setTheme] = useState<Theme>(() => {
    const saved = localStorage.getItem('theme')
    return (saved as Theme) || 'dark'
  })

  useEffect(() => {
    applyTheme(theme)
    localStorage.setItem('theme', theme)
  }, [theme])

  return { theme, setTheme }
}
```

Create `src/hooks/use-keyboard.ts`:
```typescript
import { useEffect, useCallback } from 'react'

export function useKeyboard(
  key: string,
  callback: () => void,
  options: { ctrl?: boolean; alt?: boolean; shift?: boolean } = {}
) {
  const { ctrl = false, alt = false, shift = false } = options

  const handleKeyDown = useCallback(
    (event: KeyboardEvent) => {
      const hasCtrl = ctrl ? event.ctrlKey || event.metaKey : true
      const hasAlt = alt ? event.altKey : true
      const hasShift = shift ? event.shiftKey : true

      if (event.key === key && hasCtrl && hasAlt && hasShift) {
        event.preventDefault()
        callback()
      }
    },
    [key, ctrl, alt, shift, callback]
  )

  useEffect(() => {
    window.addEventListener('keydown', handleKeyDown)
    return () => window.removeEventListener('keydown', handleKeyDown)
  }, [handleKeyDown])
}
```
